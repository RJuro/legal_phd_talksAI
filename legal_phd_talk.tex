\section{Capability vs Liability}

\begin{frame}{Capability \(\uparrow\) vs Reliability \(\leftrightarrow\)}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \textbf{The liability zone opens up}
      \begin{itemize}
        \item Frontier models swing from code assists to math olympiad-level proofs.
        \item Reliability plateaus on citation fidelity, jurisdiction discipline, and temporal accuracy.
        \item The jagged frontier: brilliant spikes next to basic failures.
        \item Deep research copilots still need accountable humans to sign outputs.
      \end{itemize}

\begin{frame}{How to work with it}
  \textbf{Treat the model as a zero-trust, occasionally brilliant wizard you must interrogate.}
  \vspace{1em}
  \begin{itemize}
    \item Require sources, intermediate steps, and explicit uncertainties.
    \item Ask: who actually knows enough to judge this output?
    \item You still sign it --- review remains non-delegable.
    \item Design workflow checklists that make validation cheap and default.
  \end{itemize}

\section{Benchmarks as Governance}

\begin{frame}{Benchmarks are power tools}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Why they matter}
      \begin{itemize}
        \item Define ``good enough'' for procurement and regulation.
        \item Shift labor and delegation once tasks go ``green'' on scoreboards.
        \item Surface liability exposure when designed around harms.
        \item Maintaining and refreshing benchmark suites becomes a sovereign capability.
      \end{itemize}
    \end{column}

\begin{frame}{Measure what matters to law}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \textbf{Performance metrics}
      \begin{itemize}
        \item MCQ accuracy, doctrinal recall, translation fidelity.
        \item Draft quality, summarisation, argument coherence.
        \item ``Bar exam'' style leaderboards and rubric scoring.
      \end{itemize}
    \end{column}
    \begin{column}{0.48\textwidth}
      \textbf{Liability metrics}
      \begin{itemize}
        \item Citation fidelity, provenance display, retrieval coverage.
        \item Jurisdiction discipline, issue spotting across circuits.
        \item Auditability, supervision hours, red-team resolution cost.
      \end{itemize}
    \end{column}
  \end{columns}
  \vfill
  \small \parencite{large_legal_fictions}
\end{frame}

\begin{frame}{Why leaderboards lie}
  \begin{columns}[T]
    \begin{column}{0.5\textwidth}
      \textbf{Contamination and drift}
      \begin{itemize}
        \item Training data leaks into public benchmarks.
        \item Static test sets invite overfitting and cherry-picking.
        \item Need for dynamic, leak-resistant evaluation sandboxes.
      \end{itemize}
